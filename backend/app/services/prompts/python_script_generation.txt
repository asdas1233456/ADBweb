# 核心规则
1. 导入: `import uiautomator2 as u2`, `import time`, `import sys`, `import io`
2. UTF-8编码: 添加编码声明和Windows兼容代码
3. 连接设备: `d = u2.connect()`
4. 步骤日志: 每步输出`print("[步骤X] 描述...")`和`print("[步骤X] 完成")`
5. 步骤截图: 每步调用`take_screenshot(d, "01_name")`
6. 异常处理: 用try-except包裹，必须完整

# 模板（直接复制使用）
```python
# -*- coding: utf-8 -*-
import uiautomator2 as u2
import time
import sys
import io

if sys.platform == 'win32':
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')
    sys.stderr = io.TextIOWrapper(sys.stderr.buffer, encoding='utf-8', errors='replace')

def take_screenshot(d, step_name):
    timestamp = int(time.time() * 1000)
    filename = f"step_{step_name}_{timestamp}.png"
    d.screenshot(filename)
    print(f"SCREENSHOT:{filename}")
    return filename

def main():
    try:
        d = u2.connect()
        print("设备连接成功")
        
        print("[步骤1] 启动应用...")
        d.app_start("com.app")
        time.sleep(2)
        print("[步骤1] 完成")
        take_screenshot(d, "01_app_started")
        
        print("[步骤2] 点击按钮...")
        d(text="按钮").click()
        time.sleep(1)
        print("[步骤2] 完成")
        take_screenshot(d, "02_button_clicked")
        
        return True
    except Exception as e:
        print(f"失败: {e}")
        try:
            take_screenshot(d, "error")
        except:
            pass
        return False

if __name__ == "__main__":
    main()
```
